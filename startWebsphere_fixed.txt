#!/bin/bash

################################################################################
# Tool Name   : startWebSphere.sh (cleaned)
# Purpose     : Start nodeagent and all JVMs listed in ck_WAS.list
# Notes       : Hostname-based JVM filtering REMOVED
################################################################################

# ======================
# CONFIGURATION
# ======================
VERSION=v90
PROFILE_NAME=node11

JVMLISTFILE=/opt/WebSphere/v90/wcc/bin/jvmrecycle/ck_WAS.list
JAVASHAREDRESOURCES="/tmp/javasharedresources"

batch=10
START_TIMEOUT=240
ret=0

# ======================
# VALIDATIONS
# ======================
if [ ! -d /opt/WebSphere/$VERSION/AppServer/profiles/$PROFILE_NAME ]; then
  echo "[FATAL] Invalid PROFILE_NAME or VERSION"
  exit 1
fi

if [ ! -s "$JVMLISTFILE" ]; then
  echo "[FATAL] JVM list file missing or empty: $JVMLISTFILE"
  exit 1
fi

# ======================
# FUNCTIONS
# ======================
serverStarted() {
  /opt/WebSphere/$VERSION/AppServer/profiles/$PROFILE_NAME/bin/serverStatus.sh "$1" \
    | grep -q "STARTED"
}

startServer() {
  local SERVER=$1

  if ps -ef | grep -w "$SERVER" | grep -v grep >/dev/null; then
    echo "WARN: $SERVER already running"
    return
  fi

  if [[ "$SERVER" == "nodeagent" ]]; then
    /opt/WebSphere/$VERSION/AppServer/profiles/$PROFILE_NAME/bin/startNode.sh \
      | grep "open for e-business"
  else
    /opt/WebSphere/$VERSION/AppServer/profiles/$PROFILE_NAME/bin/startServer.sh \
      "$SERVER" -nowait -quiet
  fi
}

# ======================
# PRE-START CLEANUP
# ======================
if [[ -d "$JAVASHAREDRESOURCES" ]]; then
  rm -f "$JAVASHAREDRESOURCES"/*semaphore* "$JAVASHAREDRESOURCES"/*memory* 2>/dev/null
fi

chmod -R g+w /opt/WebSphere/$VERSION/AppServer/profiles/$PROFILE_NAME

# ======================
# START NODEAGENT
# ======================
echo "INFO: Starting nodeagent..."
startServer nodeagent

if ! serverStarted nodeagent; then
  echo "ERROR: nodeagent failed to start"
  exit 1
fi

# ======================
# LOAD JVM LIST (ALL JVMs)
# ======================
APP_SERVER_LIST=$(egrep -v '(^[[:space:]]*#|^[[:space:]]*$)' "$JVMLISTFILE")

# ======================
# START JVMs IN BATCHES
# ======================
BATCH_LIST=""
count=0

for APP_SERVER in $APP_SERVER_LIST; do
  echo "INFO: Starting $APP_SERVER..."
  startServer "$APP_SERVER"

  BATCH_LIST="$BATCH_LIST $APP_SERVER"
  count=$((count + 1))

  if [[ $count -ge $batch ]]; then
    for JVM in $BATCH_LIST; do
      TIME_USED=0
      until serverStarted "$JVM" || [[ $TIME_USED -ge $START_TIMEOUT ]]; do
        sleep 10
        TIME_USED=$((TIME_USED + 10))
      done

      if ! serverStarted "$JVM"; then
        echo "WARN: $JVM did not reach STARTED state"
        ret=1
      fi
    done

    BATCH_LIST=""
    count=0
  fi
done

# ======================
# FINAL STATUS CHECK
# ======================
echo "INFO: Final JVM status check..."

for JVM in nodeagent $BATCH_LIST; do
  if ! serverStarted "$JVM"; then
    echo "WARN: $JVM not STARTED"
    ret=1
  fi
done

echo "INFO: startWebSphere.sh completed on $(hostname)"
exit $ret
